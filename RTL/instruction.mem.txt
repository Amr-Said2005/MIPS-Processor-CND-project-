// Instruction ROM contents -- ONE 32-BIT WORD PER LINE (read by $readmemb).
// Line N holds the instruction at byte address N*4 (the ROM is indexed pc[7:2]).
//
// Opcodes  : R_type=000000 ADDI=000001 ANDI=000010 LW=000011 SW=000100
//            BEQ=000101 J=000110 JMN=000111 SWI=001000 PMC=001001
// R funct  : add=000000 sub=000001 and=000010 or=000011 slt=000100
// Fields   : op[31:26] rs[25:21] rt[20:16] rd[15:11] shamt[10:6] funct[5:0]
// j takes a WORD address; beq offset is in words; jmn/pmc load a BYTE address
// from data memory. "trap" = addi $10,$0,99: if $10 ever becomes 99, a
// jump/branch that should have skipped it did not.
//
// Exercises ALL 14 instructions. Expected final state:
//   $1=5 $2=3 $3=8 $4=2 $5=1 $6=7 $7=0 $8=4 $9=8 $10=0
//   $11=76 $12=28 $13=104 $14=12 $15=1
//   mem[0]=8 mem[4]=76 mem[12]=104 mem[16]=100 mem[28]=7
//
// ---- arithmetic & logic ----------------------------------------------------
00000100000000010000000000000101   // w0  b0  : addi $1,$0,5     $1 = 5
00000100000000100000000000000011   // w1  b4  : addi $2,$0,3     $2 = 3
00000000001000100001100000000000   // w2  b8  : add  $3,$1,$2    $3 = 8
00000000001000100010000000000001   // w3  b12 : sub  $4,$1,$2    $4 = 2
00000000001000100010100000000010   // w4  b16 : and  $5,$1,$2    $5 = 1
00000000001000100011000000000011   // w5  b20 : or   $6,$1,$2    $6 = 7
00000000001000100011100000000100   // w6  b24 : slt  $7,$1,$2    $7 = 0 (5<3 false)
00001000001010000000000000000100   // w7  b28 : andi $8,$1,4     $8 = 5&4 = 4
// ---- memory ----------------------------------------------------------------
00010000000000110000000000000000   // w8  b32 : sw   $3,0($0)    mem[0] = 8
00001100000010010000000000000000   // w9  b36 : lw   $9,0($0)    $9 = 8
// ---- beq: not-taken, then taken --------------------------------------------
00010100001000100000000000000001   // w10 b40 : beq  $1,$2,+1    not taken (5!=3)
00010100011010010000000000000001   // w11 b44 : beq  $3,$9,+1    taken (8==8) -> w13
00000100000010100000000001100011   // w12 b48 : trap             SKIPPED
// ---- j ---------------------------------------------------------------------
00011000000000000000000000001111   // w13 b52 : j    15          -> w15 (byte 60)
00000100000010100000000001100011   // w14 b56 : trap             SKIPPED
// ---- jmn: indirect jump through memory -------------------------------------
00000100000010110000000001001100   // w15 b60 : addi $11,$0,76   $11 = 76 (= w19)
00010000000010110000000000000100   // w16 b64 : sw   $11,4($0)   mem[4] = 76
00011100000000000000000000000100   // w17 b68 : jmn  4($0)       PC = mem[4] = 76
00000100000010100000000001100011   // w18 b72 : trap             SKIPPED
// ---- swi: store then post-increment the base -------------------------------
00000100000011000000000000010100   // w19 b76 : addi $12,$0,20   $12 = 20
00100001100001100000000000001000   // w20 b80 : swi  $6,8($12)   mem[28]=7; $12=28
// ---- pmc: call via memory, save return address ------------------------------
00000100000011010000000001101000   // w21 b84 : addi $13,$0,104  $13 = 104 (= w26)
00010000000011010000000000001100   // w22 b88 : sw   $13,12($0)  mem[12] = 104
00000100000011100000000000001100   // w23 b92 : addi $14,$0,12   $14 = 12

00000100000010100000000001100011   // w25 b100: trap             SKIPPED
00000100000011110000000000000001   // w26 b104: addi $15,$0,1    $15 = 1 (landed)
